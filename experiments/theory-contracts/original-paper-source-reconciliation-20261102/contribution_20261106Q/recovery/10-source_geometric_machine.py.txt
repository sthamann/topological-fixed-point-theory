23: def profile_values(w,profile):
24:     if profile=='quadratic':return w*w,2*w
25:     # A smooth even bump, zero near w=0 and both original physical boundaries.
26:     # Its location/width are diagnostic inputs, not TFPT predictions.

37:     # In the original Pin sector containing the normal zero branch, the
38:     # allowed transverse H indices are even: E_m=pi*m, J=b.
39:     modes=np.arange(-nmax,nmax+1)
40:     nodes,weights=np.polynomial.legendre.leggauss(max(512,8*nmax))

43:     f,_=profile_values(w,profile)
44:     moments=np.cos(np.pi*orders[:,None]*w[None,:])@(weights*np.exp(-epsilon*f))
